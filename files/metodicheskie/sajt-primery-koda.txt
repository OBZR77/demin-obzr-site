4. МИНИ-ТЕСТ НА JAVASCRIPT
------------------------------------------------
<div id="quiz"></div>
<script>
const questions = [
  { q: "По какому единому номеру можно вызвать экстренные службы с мобильного телефона?",
    options: ["101", "102", "112", "103"], correct: 2 },
  { q: "[Ваш вопрос]", options: ["[А]", "[Б]", "[В]", "[Г]"], correct: 0 }
];
const box = document.getElementById("quiz");
let score = 0;
questions.forEach((item, i) => {
  const fs = document.createElement("fieldset");
  const lg = document.createElement("legend");
  lg.textContent = (i + 1) + ". " + item.q;
  fs.appendChild(lg);
  item.options.forEach((text, j) => {
    const label = document.createElement("label");
    const input = document.createElement("input");
    input.type = "radio"; input.name = "q" + i; input.value = j;
    label.append(input, " " + text);
    label.style.display = "block";
    fs.appendChild(label);
  });
  box.appendChild(fs);
});
const btn = document.createElement("button");
btn.textContent = "Проверить";
const out = document.createElement("p");
out.setAttribute("aria-live", "polite");
btn.onclick = () => {
  score = 0;
  questions.forEach((item, i) => {
    const sel = document.querySelector('input[name="q' + i + '"]:checked');
    if (sel && Number(sel.value) === item.correct) score++;
  });
  out.textContent = "Результат: " + score + " из " + questions.length;
};
box.append(btn, out);
</script>
